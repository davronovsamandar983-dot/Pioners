% ---------------------------------------------------------------------
%  Question macro — blue badge number, resets per module
% ---------------------------------------------------------------------
\newcounter{qnum}
\newcommand{\qbadge}[1]{%
  \colorbox{keycolor}{\textcolor{white}{\bfseries\strut\,#1\,}}}
% A question is an ATOMIC block: the badge, the stem, the choices and the
% work space must never be split across two columns or two pages. Wrapping
% the whole thing in a \linewidth minipage guarantees that inside multicols.
\newcommand{\question}[1]{%
  \par\vspace{9pt}%
  \stepcounter{qnum}%
  \noindent\begin{minipage}{\linewidth}%
    \qbadge{\theqnum}\par\vspace{6pt}%
    #1\par
  \end{minipage}\par
}

% Free-response question: same, with the Work Space box welded on.
