\settopicname{Linear Equations in One Variable}
\chapter{Linear Equations in One Variable}
\topicopen{Algebra}{Create, solve, and interpret a linear equation in one
variable, including equations with fractions, parentheses, and variables on
both sides, and identify when such an equation has one, no, or infinitely
many solutions.}

\begin{essentials}
Every linear equation in one variable can be reduced to the form $ax=b$.
The moves that preserve the solution set are: add or subtract the same
quantity on both sides; multiply or divide both sides by the same nonzero
number; and distribute, $a(b+c)=ab+ac$. Clear fractions first by multiplying
\emph{every} term by the least common denominator, and clear decimals by
multiplying every term by a power of $10$. Then collect the variable terms on
one side, the constants on the other, and divide by the coefficient.

For $ax=b$: if $a\neq 0$ there is exactly one solution, $x=\dfrac{b}{a}$; if
$a=0$ and $b=0$ every real number is a solution (infinitely many); if $a=0$
and $b\neq 0$ there is no solution. A literal equation is solved the same
way --- treat every other letter as a fixed number and isolate the requested
variable. In a context, the coefficient of the variable is a rate per unit
and the constant term is the fixed or starting amount.
\end{essentials}

\qfr{What value of $x$ satisfies the equation
\[ 5x+12=47\,? \]}

\question{What is the solution to the given equation?
\[ 3(x-4)=18 \]
\choices{\item $2$ \item $6$ \item $10$ \item $30$}}

\question{What value of $x$ satisfies the given equation?
\[ 7x-3=4x+9 \]
\choices{\item $2$ \item $3$ \item $4$ \item $12$}}

\qfr{What value of $x$ satisfies the equation
\[ \dfrac{x}{4}+3=11\,? \]}

\question{What is the solution to the given equation?
\[ 12-2x=30 \]
\choices{\item $-21$ \item $-9$ \item $9$ \item $21$}}

\question{What value of $x$ satisfies the given equation?
\[ \dfrac{x+9}{2}=7 \]
\choices{\item $-4$ \item $5$ \item $14$ \item $23$}}

\qfr{A gym charges a one-time registration fee of \$40 plus a monthly fee of
\$15. A member has paid a total of \$220, including the registration fee.
For how many months has the member paid?}

\question{What is the solution to the given equation?
\[ 4(2x+3)=44 \]
\choices{\item $4$ \item $\dfrac{11}{2}$ \item $7$ \item $16$}}

\question{What value of $x$ satisfies the given equation?
\[ 5(x-2)=3(x+4) \]
\choices{\item $1$ \item $7$ \item $11$ \item $22$}}

\qfr{What value of $x$ satisfies the equation
\[ \dfrac{2x-1}{3}=\dfrac{x+4}{2}\,? \]}

\question{A tank holds 250 liters of water and is drained at a constant rate
of 8 liters per minute. After how many minutes does the tank hold 90 liters
of water?
\choices{\item $11.25$ \item $20$ \item $42.5$ \item $160$}}

\question{What value of $x$ satisfies the given equation?
\[ 0.4x+1.2=0.9x-2.3 \]
\choices{\item $-7$ \item $1.75$ \item $2.2$ \item $7$}}

\question{The equation $P=2\ell+2w$ gives the perimeter $P$ of a rectangle
with length $\ell$ and width $w$. Which equation correctly expresses $w$ in
terms of $P$ and $\ell$?
\choices{\item $w=\dfrac{P-2\ell}{2}$ \item $w=\dfrac{P}{2}-2\ell$
\item $w=P-2\ell$ \item $w=\dfrac{P-\ell}{2}$}}

\qfr{What value of $x$ satisfies the equation
\[ 3(2x-5)-4(x-3)=11\,? \]}

\question{A phone plan costs \$25 per month plus \$0.10 for each text
message sent. In a month in which the total charge was \$43, how many text
messages were sent?
\choices{\item $18$ \item $180$ \item $430$ \item $680$}}

\question{The equation $C=8n+150$ gives the total cost $C$, in dollars, of
producing $n$ items at a workshop. What is the best interpretation of $8$ in
this context?
\choices{\item The fixed cost, in dollars, before any items are produced.
\item The cost, in dollars, to produce each additional item.
\item The total cost, in dollars, of producing 8 items.
\item The number of items that can be produced for 150 dollars.}}

\qfr{What value of $x$ satisfies the equation
\[ \dfrac{x}{2}-\dfrac{x}{5}=6\,? \]}

\question{In the given equation, $k$ is a constant.
\[ 3(2x+5)=6x+k \]
If the equation has infinitely many solutions, what is the value of $k$?
\choices{\item $-15$ \item $3$ \item $5$ \item $15$}}

\question{In the given equation, $a$ is a constant.
\[ ax-12=4x+9 \]
For what value of $a$ does the equation have no solution?
\choices{\item $-4$ \item $0$ \item $4$ \item $9$}}

\qfr{What value of $x$ satisfies the equation
\[ 2(3x-4)+7=3(x+8)-2x\,? \]}

\question{Which equation correctly expresses $x$ in terms of $y$ for
\[ y=\dfrac{3x-5}{4}\,? \]
\choices{\item $x=\dfrac{4y+5}{3}$ \item $x=\dfrac{4y-5}{3}$
\item $x=\dfrac{y+5}{3}$ \item $x=4y+\dfrac{5}{3}$}}

\question{In the given equation, $c$ is a constant.
\[ 5x+c=5x-8 \]
Which statement about the given equation must be true?
\choices{\item The equation has exactly one solution for every value of c.
\item The equation has no solution for every value of c.
\item The equation has infinitely many solutions when c equals negative 8,
and no solution otherwise.
\item The equation has infinitely many solutions for every value of c.}}

\end{multicols*}
