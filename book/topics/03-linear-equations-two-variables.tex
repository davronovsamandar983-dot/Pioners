\settopicname{Linear Equations in Two Variables}
\chapter{Linear Equations in Two Variables}
\topicopen{Algebra}{Create, use, and interpret a linear equation in two
variables: write the line from a graph, from two points, or from a
description, and read its slope and intercepts back out in context.}

\begin{essentials}
A line in the $xy$-plane can be written three ways.
\emph{Slope-intercept:} $y=mx+b$, slope $m$, $y$-intercept $(0,b)$.
\emph{Point-slope:} $y-y_1=m(x-x_1)$ for the point $(x_1,y_1)$.
\emph{Standard:} $ax+by=c$, slope $-\dfrac{a}{b}$, $x$-intercept
$\left(\dfrac{c}{a},0\right)$, $y$-intercept $\left(0,\dfrac{c}{b}\right)$.
From two points,
\[ m=\frac{y_2-y_1}{x_2-x_1}\quad(\text{rise over run}), \]
subtracting in the same order top and bottom. Set $y=0$ for the
$x$-intercept and $x=0$ for the $y$-intercept. Parallel lines have equal
slopes; perpendicular lines have slopes $m$ and $-\dfrac{1}{m}$ --- the
negative reciprocal, not the reciprocal. Two equations graph the same line
when one is a nonzero multiple of the other. In a context $ax+by=c$, the
coefficients $a$ and $b$ are costs or rates per unit, $c$ is the fixed
total, and each intercept is the case in which the other quantity is $0$.
\end{essentials}

\modulehead{Module 1}{Questions 1--22}
\begin{multicols*}{2}

\question{The graph of the equation $y=4x-9$ is a line in the $xy$-plane.
What is the slope of the line?
\choices{\item $-9$ \item $-4$ \item $4$ \item $9$}}

\qfr{The line with equation $y=-3x+11$ is graphed in the $xy$-plane. What is
the $y$-coordinate of the $y$-intercept of this line?}

\question{A line in the $xy$-plane passes through the points $(2,3)$ and
$(6,11)$. What is the slope of the line?
\choices{\item $-2$ \item $\dfrac{1}{2}$ \item $\dfrac{7}{4}$ \item $2$}}

\question{The graph of $2x+5y=20$ is a line in the $xy$-plane. What is the
$x$-coordinate of the $x$-intercept of the line?
\choices{\item $\dfrac{5}{2}$ \item $4$ \item $10$ \item $20$}}

\qfr{A line in the $xy$-plane has slope $3$ and passes through the point
$(0,-4)$. What is the value of $y$ when $x=6$?}

\question{A line in the $xy$-plane has slope $-2$ and $y$-intercept $(0,7)$.
Which equation represents this line?
\choices{\item $y=-2x+7$ \item $y=-2x-7$ \item $y=2x+7$ \item $y=7x-2$}}

\qfr{The equation $4x+y=19$ represents a line in the $xy$-plane. What is the
value of $y$ when $x=3$?}

\question{What is the slope of the line with equation $3x+6y=18$ in the
$xy$-plane?
\choices{\item $-2$ \item $-\dfrac{1}{2}$ \item $\dfrac{1}{2}$ \item $3$}}

\question{A line in the $xy$-plane passes through the point $(4,-1)$ and has
slope $\dfrac{1}{2}$. Which equation represents the line?
\choices{\item $y=\dfrac{1}{2}x-1$ \item $y=\dfrac{1}{2}x+1$
\item $y=\dfrac{1}{2}x-3$ \item $y=2x-9$}}

\qfr{A line in the $xy$-plane passes through the points $(-3,5)$ and
$(7,k)$, where $k$ is a constant. The slope of the line is $-2$. What is the
value of $k$?}

\question{Line $\ell$ has equation $y=-\dfrac{3}{4}x+2$. Line $m$ is parallel
to line $\ell$ and passes through the point $(8,1)$. Which equation
represents line $m$?
\choices{\item $y=-\dfrac{3}{4}x-5$ \item $y=-\dfrac{3}{4}x+2$
\item $y=-\dfrac{3}{4}x+7$ \item $y=\dfrac{4}{3}x-\dfrac{29}{3}$}}

\question{In the $xy$-plane, line $k$ is perpendicular to the line with
equation $y=\dfrac{2}{5}x-3$. What is the slope of line $k$?
\choices{\item $-\dfrac{5}{2}$ \item $-\dfrac{2}{5}$ \item $\dfrac{2}{5}$
\item $\dfrac{5}{2}$}}

\qfr{A tank holds $60$ liters of water. Water drains from the tank at a
constant rate of $4$ liters per minute, so the volume $V$, in liters, of
water left in the tank $t$ minutes after draining begins is $V=60-4t$. After
how many minutes does the tank hold $24$ liters of water?}

\question{The total cost $C$, in dollars, of renting a floor sander for $h$
hours is given by $C=12h+35$. What is the best interpretation of $12$ in
this context?
\choices{\item The one-time fee, in dollars, charged in addition to the
hourly rate.
\item The total cost, in dollars, of renting the sander for $35$ hours.
\item The number of hours the sander is rented.
\item The increase in the total cost, in dollars, for each additional hour
of rental.}}

\qfr{A line in the $xy$-plane passes through the points $(-2,7)$ and
$(4,-5)$. What is the $y$-coordinate of the $y$-intercept of the line?}

\question{A grocery order consists of $x$ pounds of rice and $y$ pounds of
beans. Rice costs \$2 per pound, beans cost \$3 per pound, and the order
costs \$36 in total, so $2x+3y=36$. What is the best interpretation of the
$x$-intercept of the graph of this equation in the $xy$-plane?
\choices{\item The order contains $12$ pounds of beans and no rice.
\item The order contains $18$ pounds of rice and no beans.
\item The order contains $18$ pounds of beans and no rice.
\item The order costs $18$ dollars in total.}}

\question{The graph of a line in the $xy$-plane is shown.
\begin{center}
\begin{tikzpicture}[scale=0.40, line width=0.7pt]
  \draw[gray!30] (-1,-1) grid (9,5);
  \draw[->] (-1.2,0) -- (9.4,0) node[below,font=\scriptsize]{$x$};
  \draw[->] (0,-1.2) -- (0,5.4) node[left,font=\scriptsize]{$y$};
  \draw[keycolor, line width=1pt] (-0.8,4.4) -- (9.2,-0.6);
  \fill (0,4) circle (0.13);
  \node[anchor=west,font=\scriptsize] at (0.35,4.6) {$(0,4)$};
  \fill (8,0) circle (0.13);
  \node[anchor=north,font=\scriptsize] at (7.8,-0.35) {$(8,0)$};
\end{tikzpicture}
\end{center}
Which equation represents the line?
\choices{\item $y=-2x+4$ \item $y=-\dfrac{1}{2}x+8$
\item $y=\dfrac{1}{2}x+4$ \item $y=-\dfrac{1}{2}x+4$}}

\question{In the $xy$-plane, the graph of $kx+3y=12$ passes through the point
$(2,-4)$, where $k$ is a constant. What is the value of $k$?
\choices{\item $-\dfrac{3}{2}$ \item $0$ \item $6$ \item $12$}}

\qfr{Line $m$ is perpendicular to the line with equation
$y=\dfrac{1}{3}x+5$ and passes through the point $(6,2)$. What is the
$y$-coordinate of the $y$-intercept of line $m$?}

\question{In the $xy$-plane, line $\ell$ has equation $ax+by=c$, where $a$,
$b$, and $c$ are positive constants. Which statement must be true?
\choices{\item The slope of line $\ell$ is negative.
\item The slope of line $\ell$ is positive.
\item Line $\ell$ passes through the origin.
\item The $x$-intercept of line $\ell$ is negative.}}

\question{Which equation correctly expresses $x$ in terms of $y$ for
\[ 5x-2y=30\,? \]
\choices{\item $x=\dfrac{2y-30}{5}$ \item $x=\dfrac{2y+30}{5}$
\item $x=\dfrac{5y+30}{2}$ \item $x=\dfrac{2y}{5}+30$}}

\question{In the $xy$-plane, line $\ell$ has $x$-intercept $(-4,0)$ and
$y$-intercept $(0,6)$. Line $n$ is perpendicular to line $\ell$ and passes
through the origin. Which equation represents line $n$?
\choices{\item $y=-\dfrac{3}{2}x$ \item $y=-\dfrac{2}{3}x$
\item $y=\dfrac{2}{3}x$ \item $y=\dfrac{3}{2}x$}}
