\chapter{Equivalent Expressions}
\topicopen{Advanced Math}{Rewrite a polynomial or rational expression in an
equivalent form, and recognise which of four given expressions is identically
equal to a given one.}

\begin{essentials}
Two expressions are equivalent when they give the same value for every input
both accept. Expanding multiplies every term by every term and then collects
like terms; factoring reverses it. Know on sight
\[ a^2-b^2=(a-b)(a+b),\qquad (a\pm b)^2=a^2\pm 2ab+b^2, \]
\[ a^3-b^3=(a-b)(a^2+ab+b^2). \]
To factor $x^2+bx+c$, find two numbers whose product is $c$ and whose sum is
$b$. For $ax^2+bx+c$, split $bx$ using two numbers whose product is $ac$ and
whose sum is $b$, then group. With rational expressions, factor the numerator
and the denominator completely and cancel only whole common \emph{factors},
never one term of a sum. Add or subtract only over a common denominator, and
divide by multiplying by the reciprocal. Clear a complex fraction by
multiplying its top and bottom by the common denominator of the small
fractions. If $d(x)$ divides $p(x)$ exactly, the quotient is the other factor.
\end{essentials}

\modulehead{Module 1}{Questions 1--22}
\begin{multicols*}{2}

\question{Which expression is equivalent to
\[ 3x(x+4)\,? \]
\choices{\item $3x^2+4$ \item $3x^2+12x$ \item $3x^2+4x$ \item $3x+12x$}}

\question{Which expression is equivalent to
\[ (x+7)(x-7)\,? \]
\choices{\item $x^2-49$ \item $x^2+49$ \item $x^2-14x-49$
\item $x^2-14x+49$}}

\qfr{The expression $(x+3)(x+5)$ is equivalent to $x^2+bx+15$, where $b$ is a
constant. What is the value of $b$?}

\question{Which expression is equivalent to
\[ 4x^2-20x\,? \]
\choices{\item $4x(x-20)$ \item $4x(x+5)$ \item $2x(2x-20)$
\item $4x(x-5)$}}

\qfr{The expression $x^2-16$ is equivalent to $(x+4)(x+k)$, where $k$ is a
constant. What is the value of $k$?}

\question{Which expression is equivalent to
\[ (3x^2+2x-5)+(x^2-6x+1)\,? \]
\choices{\item $4x^2-4x-6$ \item $4x^2+8x-4$ \item $3x^2-4x-4$
\item $4x^2-4x-4$}}

\qfr{The expression $(3x)(4x^2)$ is equivalent to $ax^3$, where $a$ is a
constant. What is the value of $a$?}

\question{Which expression is equivalent to
\[ x^2+9x+20\,? \]
\choices{\item $(x+2)(x+10)$ \item $(x-4)(x-5)$ \item $(x+4)(x+5)$
\item $(x+1)(x+20)$}}

\question{Which expression is equivalent to
\[ (2x-5)(3x+4)\,? \]
\choices{\item $6x^2-7x-20$ \item $6x^2+7x-20$ \item $6x^2-23x-20$
\item $6x^2-7x+20$}}

\qfr{The expression $16x^2-c$, where $c$ is a positive constant, is equivalent
to $(4x-9)(4x+9)$. What is the value of $c$?}

\question{Which expression is equivalent to
\[ \dfrac{x^2-9}{x+3}\,, \]
where $x\neq -3$?
\choices{\item $x+3$ \item $x-3$ \item $x^2-3$ \item $x-9$}}

\qfr{For $x\neq -3$, the expression
\[ \dfrac{x^2+7x+12}{x+3} \]
is equivalent to $x+k$, where $k$ is a constant. What is the value of $k$?}

\question{Which expression is equivalent to
\[ (x+5)^2-(x-5)^2\,? \]
\choices{\item $50$ \item $20x$ \item $10x$ \item $2x^2+50$}}

\question{Which expression is equivalent to
\[ 2x^2+7x+3\,? \]
\choices{\item $(2x+1)(x+3)$ \item $(2x+3)(x+1)$ \item $(2x-1)(x-3)$
\item $(2x+7)(x+3)$}}

\qfr{For $x\neq -3$, the expression
\[ \dfrac{2x^2+5x-3}{x+3} \]
is equivalent to $2x+k$, where $k$ is a constant. What is the value of $k$?}

\question{Which expression is equivalent to
\[ (5x^2-3x+2)-(2x^2+4x-7)\,? \]
\choices{\item $3x^2-7x-5$ \item $3x^2+x-5$ \item $3x^2-7x+9$
\item $7x^2+x+9$}}

\question{Which expression is equivalent to
\[ \dfrac{1}{x}+\dfrac{1}{2x}\,, \]
where $x\neq 0$?
\choices{\item $\dfrac{1}{3x}$ \item $\dfrac{2}{3x}$
\item $\dfrac{3}{2x^2}$ \item $\dfrac{3}{2x}$}}

\question{Which expression is equivalent to
\[ \dfrac{3x^2+11x-4}{x+4}\,, \]
where $x\neq -4$?
\choices{\item $3x-1$ \item $3x+1$ \item $3x-4$ \item $x-1$}}

\question{The expression $(x+a)(x+3)$, where $a$ is a constant, is equivalent
to $x^2+10x+b$, where $b$ is a constant. What is the value of $b$?
\choices{\item $7$ \item $13$ \item $21$ \item $30$}}

\question{Which expression is equivalent to
\[ \dfrac{1}{x-2}-\dfrac{1}{x+2}\,, \]
where $x\neq \pm 2$?
\choices{\item $\dfrac{2x}{x^2-4}$ \item $\dfrac{4}{x^2+4}$
\item $\dfrac{-4}{x^2-4}$ \item $\dfrac{4}{x^2-4}$}}

\qfr{In the expression $x^2+kx+36$, $k$ is a constant. If the expression is
equivalent to $(x+6)^2$, what is the value of $k$?}

\question{Which expression is equivalent to
\[ \dfrac{x^2-5x+6}{x^2-9}\,, \]
where $x\neq \pm 3$?
\choices{\item $\dfrac{x-2}{x-3}$ \item $\dfrac{x+2}{x+3}$
\item $\dfrac{x-2}{x+3}$ \item $\dfrac{x-3}{x+3}$}}

\end{multicols*}

\item $x^3-2x^2+5x+20$ \item $x^3+6x^2-3x+20$}}

\qfr{The expression $(2x-3)^2$ is equivalent to $4x^2+bx+9$, where $b$ is a
constant. What is the value of $b$?}

\question{Which expression is equivalent to

\choices{\item $3x(2x-3)^2$ \item $3x(2x+3)(2x-3)$ \item $3x(4x^2-9x)$
\item $9x(x^2-3)$}}

\question{Which expression is equivalent to
\[ \dfrac{6x^2+9x}{3x}\,, \]
where $x\neq 0$?
\choices{\item $2x+9$ \item $2x^2+3x$ \item $6x^2+3$ \item $2x+3$}}

\question{Which expression is equivalent to
\[ \dfrac{x^2-4}{x-2}+3\,, \]
where $x\neq 2$?
\choices{\item $x+1$ \item $x+2$ \item $x+5$ \item $3x+6$}}

\question{Which expression is equivalent to
\[ \dfrac{x^2+6x+9}{x^2-9}\,, \]
where $x\neq \pm 3$?
\choices{\item $\dfrac{1}{x-3}$ \item $\dfrac{x+3}{x-3}$
\item $\dfrac{x-3}{x+3}$ \item $\dfrac{-(x+3)}{x-3}$}}

\qfr{For $x\neq \tfrac{5}{2}$, the expression
\[ \dfrac{6x^2-7x-20}{2x-5} \]
is equivalent to $3x+k$, where $k$ is a constant. What is the value of $k$?}

\question{Which expression is equivalent to
\[ \dfrac{x}{x+1}\cdot\dfrac{x^2-1}{x}\,, \]
where $x\neq 0$ and $x\neq -1$?
\choices{\item $x+1$ \item $x-1$ \item $x^2-1$ \item $\dfrac{x-1}{x}$}}

\question{Which expression is equivalent to
\[ \dfrac{\ \dfrac{1}{x}-\dfrac{1}{4}\ }{x-4}\,, \]
where $x\neq 0$ and $x\neq 4$?
\choices{\item $\dfrac{4-x}{4x}$ \item $\dfrac{1}{4x}$
\item $\dfrac{-1}{4x-16}$ \item $\dfrac{-1}{4x}$}}

\qfr{The expression $(x+5)(x-2)-(x^2-3x)$ is equivalent to $ax-10$, where $a$
is a constant. What is the value of $a$?}

\question{Which expression is equivalent to
\[ \dfrac{2}{x-3}+\dfrac{5}{x+1}\,, \]
where $x\neq 3$ and $x\neq -1$?
\choices{\item $\dfrac{7}{x^2-2x-3}$ \item $\dfrac{7x+13}{x^2-2x-3}$
\item $\dfrac{7x-13}{x^2-2x-3}$ \item $\dfrac{7x-13}{x^2+2x-3}$}}

\question{Which expression is equivalent to
\[ 25x^2-30x+9\,? \]
\choices{\item $(5x-3)^2$ \item $(5x+3)^2$ \item $(5x-9)(5x-1)$
\item $(25x-9)(x-1)$}}

\question{Which expression is equivalent to
\[ (x^2+2x-1)(x-3)\,? \]
\choices{\item $x^3+5x^2-7x+3$ \item $x^3-x^2-7x-3$ \item $x^3-x^2+5x+3$
\item $x^3-x^2-7x+3$}}

\question{Which expression is equivalent to
\[ \dfrac{x^2-2x-15}{x^2-25}\,, \]
where $x\neq \pm 5$?
\choices{\item $\dfrac{x-3}{x-5}$ \item $\dfrac{x+3}{x-5}$
\item $\dfrac{x-3}{x+5}$ \item $\dfrac{x+3}{x+5}$}}

\qfr{For $x\neq -1$, the expression
\[ \dfrac{x^3+2x^2-5x-6}{x+1} \]
is equivalent to $x^2+x+c$, where $c$ is a constant. What is the value of
$c$?}

\question{Which expression is equivalent to
\[ \dfrac{x^2+5x+6}{x+2}\cdot\dfrac{x^2-4}{x^2-9}\,, \]
where $x\neq -2$ and $x\neq \pm 3$?
\choices{\item $\dfrac{x^2-4}{x-3}$ \item $\dfrac{x^2-4}{x+3}$
\item $\dfrac{x+2}{x-3}$ \item $\dfrac{x-2}{x-3}$}}

\qfr{In the equation $(ax+3)(2x-5)=8x^2+bx-15$, $a$ and $b$ are constants and
the equation is true for all values of $x$. What is the value of $b$?}

\question{Which expression is equivalent to
\[ \dfrac{1}{\ \dfrac{1}{x}+\dfrac{1}{y}\ }\,, \]
where $x$, $y$, and $x+y$ are all nonzero?
\choices{\item $\dfrac{x+y}{xy}$ \item $x+y$ \item $\dfrac{xy}{x+y}$
\item $\dfrac{1}{x+y}$}}

\qfr{In the given expression, $k$ is a constant. For $x\neq \tfrac{4}{3}$,
\[ \dfrac{3x^2+kx-8}{3x-4} \]
is equivalent to $x+2$. What is the value of $k$?}

\question{Which expression is equivalent to
\[ \dfrac{x^3-8}{x-2}\,, \]
where $x\neq 2$?
\choices{\item $x^2-2x+4$ \item $x^2+2x+4$ \item $x^2+4$ \item $x^2-4$}}

\qfr{For $x\neq \pm\tfrac{3}{2}$, the expression
\[ \dfrac{4x^2-9}{2x+3}+\dfrac{4x^2-9}{2x-3} \]
is equivalent to $ax$, where $a$ is a constant. What is the value of $a$?}

\question{Which expression is equivalent to
\[ \dfrac{2x}{x^2-1}-\dfrac{1}{x-1}\,, \]
where $x\neq \pm 1$?
\choices{\item $\dfrac{1}{x-1}$ \item $\dfrac{2x-1}{x^2-1}$
\item $\dfrac{1}{x+1}$ \item $\dfrac{-1}{x+1}$}}

\end{multicols*}
