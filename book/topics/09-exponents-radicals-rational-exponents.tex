\chapter{Exponents, Radicals, and Rational Exponents}
\topicopen{Advanced Math}{Rewrite an expression involving integer or rational
exponents, radicals, or scientific notation in an equivalent form, and solve
an exponential equation by writing both sides over a common base.}

\begin{essentials}
For a nonzero base, $a^{m}a^{n}=a^{m+n}$, $\dfrac{a^{m}}{a^{n}}=a^{m-n}$,
$(a^{m})^{n}=a^{mn}$, $(ab)^{n}=a^{n}b^{n}$, $a^{0}=1$, and
$a^{-n}=\dfrac{1}{a^{n}}$. A factor moved across the fraction bar changes the
sign of its exponent. Rational exponents are radicals:
$a^{1/n}=\sqrt[n]{a}$ and $a^{m/n}=\sqrt[n]{a^{m}}=\left(\sqrt[n]{a}\right)^{m}$,
and every law above still applies to them. Simplify a radical by pulling out
perfect-power factors, as in $\sqrt{50}=\sqrt{25\cdot 2}=5\sqrt{2}$; radicals
add only when the radicands match, as in $2\sqrt{3}+5\sqrt{3}=7\sqrt{3}$. To
rationalise a denominator, multiply above and below by $\sqrt{a}$, or by the
conjugate when the denominator is $p\pm\sqrt{q}$. To solve $b^{f(x)}=b^{g(x)}$,
rewrite both sides over one base and set the exponents equal. In scientific
notation, $(a\times 10^{m})(c\times 10^{n})=ac\times 10^{m+n}$; then adjust so
the leading factor is at least $1$ and less than $10$.
\end{essentials}

\modulehead{Module 1}{Questions 1--22}
\begin{multicols*}{2}

\question{Which expression is equivalent to
\[ x^{4}\cdot x^{7}\,? \]
\choices{\item $x^{28}$ \item $x^{11}$ \item $x^{3}$ \item $2x^{11}$}}

\qfr{What is the value of $7^{0}+2^{3}$?}

\question{Which expression is equivalent to
\[ \dfrac{x^{9}}{x^{4}}\,? \]
\choices{\item $x^{5}$ \item $x^{13}$ \item $x^{36}$ \item $x^{9/4}$}}

\qfr{What is the value of $4^{-2}$?}

\question{Which expression is equivalent to
\[ \left(x^{3}\right)^{5}? \]
\choices{\item $x^{8}$ \item $3x^{5}$ \item $x^{15}$ \item $x^{-2}$}}

\qfr{What is the value of $\sqrt{49}+\sqrt{16}$?}

\question{Which expression is equivalent to
\[ \sqrt[3]{x}\,? \]
\choices{\item $x^{3}$ \item $3x$ \item $x^{-3}$ \item $x^{1/3}$}}

\question{Which expression is equivalent to
\[ \left(3\times 10^{4}\right)\left(2\times 10^{3}\right)? \]
\choices{\item $6\times 10^{12}$ \item $5\times 10^{7}$
\item $6\times 10^{8}$ \item $6\times 10^{7}$}}

\qfr{The expression $\sqrt{48}$ can be written in the form $a\sqrt{3}$, where
$a$ is a positive integer. What is the value of $a$?}

\question{Which expression is equivalent to
\[ \left(2x^{3}\right)^{4}? \]
\choices{\item $8x^{12}$ \item $16x^{12}$ \item $16x^{7}$ \item $2x^{12}$}}

\question{Which expression is equivalent to
\[ \sqrt{50}\,? \]
\choices{\item $5\sqrt{2}$ \item $2\sqrt{5}$ \item $25\sqrt{2}$
\item $10\sqrt{5}$}}

\question{What value of $x$ satisfies the equation
\[ 2^{\,x+1}=32\,? \]
\choices{\item $5$ \item $16$ \item $4$ \item $31$}}

\qfr{What is the value of $9^{3/2}$?}

\question{Which expression is equivalent to
\[ \dfrac{6}{\sqrt{3}}\,? \]
\choices{\item $3\sqrt{2}$ \item $2\sqrt{3}$ \item $6\sqrt{3}$
\item $\dfrac{\sqrt{3}}{2}$}}

\question{For positive $x$ and $y$, which expression is equivalent to
\[ \dfrac{3x^{-2}}{y^{-3}}\,? \]
\choices{\item $\dfrac{3x^{2}}{y^{3}}$ \item $\dfrac{y^{3}}{3x^{2}}$
\item $-\dfrac{3y^{3}}{x^{2}}$ \item $\dfrac{3y^{3}}{x^{2}}$}}

\question{One grain of a powder has a mass of $2.5\times 10^{-5}$ grams. What
is the total mass, in grams, of $4\times 10^{6}$ grains of the powder?
\choices{\item $100$ \item $10$ \item $1{,}000$ \item $0.1$}}

\qfr{What is the value of $64^{2/3}-49^{1/2}$?}

\question{What value of $x$ satisfies the equation
\[ 3^{\,2x}=27^{\,x-1}? \]
\choices{\item $-3$ \item $1$ \item $3$ \item $\dfrac{1}{3}$}}

\question{For $x>0$, which expression is equivalent to
\[ \dfrac{\sqrt[3]{x^{5}}}{\sqrt{x}}\,? \]
\choices{\item $x^{13/6}$ \item $x^{7/6}$ \item $x^{5/6}$ \item $x^{10/3}$}}

\question{Which expression is equivalent to
\[ \dfrac{4}{\sqrt{5}-1}\,? \]
\choices{\item $\sqrt{5}-1$ \item $\dfrac{\sqrt{5}+1}{4}$
\item $4\sqrt{5}+4$ \item $\sqrt{5}+1$}}

\qfr{What value of $x$ satisfies the equation
\[ \left(\dfrac{1}{2}\right)^{x}=8^{\,x-5}? \]}

\question{For positive $a$ and $b$, which expression is equivalent to
\[ \dfrac{a^{-3}b^{5}}{a^{2}b^{-1}}\,? \]
\choices{\item $\dfrac{b^{6}}{a^{5}}$ \item $\dfrac{b^{4}}{a^{5}}$
\item $\dfrac{b^{6}}{a}$ \item $\dfrac{a^{5}}{b^{6}}$}}

\end{multicols*}
