\chapter{Linear Inequalities in One or Two Variables}
\topicopen{Algebra}{Create, solve, and interpret linear inequalities in one or
two variables, including systems of inequalities used to model the
constraints of a real-world situation.}

\begin{essentials}
A linear inequality is solved with the same moves as a linear equation --- add
or subtract the same quantity, multiply or divide by the same nonzero number
--- with one extra rule: multiplying or dividing both sides by a \textbf{negative}
number \textbf{reverses} the direction of the sign.
\[ -2x < 6 \iff x > -3 \]
A compound inequality $a < bx + c \le d$ is two inequalities at once; every
move must be applied to all three parts, and a reversal flips the whole chain.

In context, \emph{at most} and \emph{no more than} mean $\le$; \emph{at least}
and \emph{no less than} mean $\ge$; \emph{fewer than} means $<$. A quantity
counted in whole units must also be a nonnegative integer, so a boundary such
as $n \le 6.7$ is rounded inward to $n = 6$.

An ordered pair $(x,y)$ satisfies a system of inequalities only when it
satisfies \textbf{every} inequality in the system. Two inequalities whose
boundary lines have the same slope but incompatible constants have no
solution at all.
\end{essentials}

\modulehead{Module 1}{Questions 1--22}
\begin{multicols*}{2}

\question{Which inequality represents all solutions of $x + 9 \le 15\,$?
\choices{\item $x \le 6$ \item $x \ge 6$ \item $x \le 24$ \item $x \ge 24$}}

\qfr{What is the greatest value of $x$ that satisfies $5x \le 40\,$?}

\question{Which inequality represents all solutions of $-3x > 12\,$?
\choices{\item $x < -4$ \item $x > -4$ \item $x < 4$ \item $x > 4$}}

\question{What is the least integer value of $x$ that satisfies
$2x - 3 > 7\,$?
\choices{\item $4$ \item $5$ \item $6$ \item $7$}}

\question{A student has \$40 to spend on notebooks that cost \$6 each. What
is the greatest number of notebooks the student can buy?
\choices{\item $5$ \item $6$ \item $7$ \item $8$}}

\qfr{What is the least value of $x$ that satisfies
$\dfrac{x}{4} + 3 \ge 7\,$?}

\question{Which value of $x$ is a solution of $-2 < x + 1 \le 5\,$?
\choices{\item $-4$ \item $-3$ \item $4$ \item $5$}}

\question{What is the least value of $x$ that satisfies
$3x + 2 \le 5x - 8\,$?
\choices{\item $3$ \item $4$ \item $5$ \item $6$}}

\question{What is the greatest integer value of $x$ that satisfies
$4(x - 3) < 2x + 6\,$?
\choices{\item $7$ \item $8$ \item $9$ \item $10$}}

\qfr{An empty delivery van has a mass of $1{,}200$ kilograms. Each crate
loaded into the van has a mass of $45$ kilograms. The loaded van must have a
mass of no more than $2{,}325$ kilograms. What is the greatest number of
crates that can be loaded?}

\question{How many integer values of $x$ satisfy $-7 \le 2x - 1 < 9\,$?
\choices{\item $7$ \item $8$ \item $9$ \item $10$}}

\question{Which inequality represents all solutions of
\[ -\dfrac{x}{3} + 2 \ge 5\,? \]
\choices{\item $x \le -9$ \item $x \ge -9$ \item $x \le 9$ \item $x \ge 9$}}

\qfr{What is the least value of $x$ that satisfies
\[ \dfrac{x}{4} + 12 \le \dfrac{x}{2} - 3\,? \]}

\question{A phone plan costs \$25 per month plus \$0.10 for each minute of
calling. A customer will pay no more than \$60 for one month. What is the
greatest number of minutes of calling the customer can use in that month?
\choices{\item $300$ \item $350$ \item $400$ \item $600$}}

\question{Which ordered pair $(x,y)$ is a solution to the system of
inequalities
\[ \begin{cases} y > 2x - 1 \\ y \le -x + 5 \end{cases} \]
\choices{\item $(3,2)$ \item $(0,0)$ \item $(1,5)$ \item $(4,1)$}}

\question{A technician charges a service fee of \$45 plus \$15 for each hour
of work. A customer will spend at most \$300, so $15h + 45 \le 300$, where
$h$ is the number of hours of work. What is the best interpretation of $15$
in this context?
\choices{\item The technician charges \$15 for each hour of work.
\item The technician charges a one-time fee of \$15.
\item The technician works at most $15$ hours.
\item The customer will spend at most \$15.}}

\qfr{What is the greatest value of $x$ that satisfies
$5 - 2x \ge x - 7\,$?}

\question{In the given inequality, $k$ is a constant.
\[ 3x + k > 12 \]
The solutions of the inequality are given by $x > 5$. What is the value of
$k$?
\choices{\item $-27$ \item $-3$ \item $3$ \item $27$}}

\question{In the given system of inequalities, $c$ is a constant.
\[ \begin{cases} y \ge 4x + 7 \\ y \le 4x + c \end{cases} \]
The system has no solution. What is the greatest integer value of $c$?
\choices{\item $5$ \item $6$ \item $7$ \item $8$}}

\qfr{How many integer values of $x$ satisfy $-4 \le 3 - 2x < 9\,$?}

\question{A pair of numbers $(x,y)$ satisfies both $3x + 2y \le 30$ and
$x + y \ge 12$. If $x = 4$, what is the greatest possible value of $y$?
\choices{\item $8$ \item $9$ \item $10$ \item $12$}}

\qfr{In the given inequality, $a$ is a constant.
\[ ax + 8 \le 3x + 20 \]
The solutions of the inequality are given by $x \le 6$. What is the value of
$a$?}

\end{multicols*}

\modulehead{Module 2}{Questions 1--22}
\begin{multicols*}{2}

\question{What is the greatest integer value of $x$ that satisfies
$2(3 - x) > 4x - 6\,$?
\choices{\item $1$ \item $2$ \item $3$ \item $4$}}

\question{Which inequality is equivalent to
\[ -\dfrac{2}{3}x + 4 \le 10\,? \]
\choices{\item $x \ge -9$ \item $x \le -9$ \item $x \ge 9$ \item $x \le 9$}}

\qfr{A temperature of $t$ degrees Celsius corresponds to $\frac{9}{5}t + 32$
degrees Fahrenheit. A storage room must be kept between $68$ and $122$
degrees Fahrenheit, inclusive. What is the greatest value of $t$, in degrees
Celsius, allowed in the storage room?}

\question{Which ordered pair $(x,y)$ is a solution to the system of
inequalities
\[ \begin{cases} y < -2x + 6 \\ y \ge x - 3 \end{cases} \]
\choices{\item $(5,1)$ \item $(1,2)$ \item $(0,-4)$ \item $(4,1)$}}

\question{A ride service charges \$2.50 plus \$1.75 for each mile traveled. A
rider will spend at most \$30 on one ride. What is the greatest whole number
of miles the rider can travel on that ride?
\choices{\item $14$ \item $15$ \item $16$ \item $17$}}

\qfr{What is the greatest integer value of $x$ that satisfies
\[ -1 \le \dfrac{3x - 4}{2} < 7\,? \]}

\question{In the given inequality, $k$ is a constant.
\[ kx - 9 < 6 \]
The solutions of the inequality are given by $x < 5$. What is the value of
$k$?
\choices{\item $-3$ \item $\dfrac{1}{3}$ \item $3$ \item $5$}}

\question{A caterer fills small trays that each hold $8$ servings and large
trays that each hold $20$ servings. She fills a total of at most $15$ trays
and needs at least $200$ servings. Which system of inequalities represents
this situation, where $s$ is the number of small trays and $\ell$ is the
number of large trays?
\choices{\item $s + \ell \le 15$ and $8s + 20\ell \ge 200$
\item $s + \ell \ge 15$ and $8s + 20\ell \le 200$
\item $s + \ell \le 15$ and $8s + 20\ell \le 200$
\item $8s + 20\ell \le 15$ and $s + \ell \ge 200$}}

\qfr{A pair of numbers $(x,y)$ satisfies $12x + 7y \le 84$. If $x = 3$ and
$y$ is a positive integer, what is the greatest possible value of $y$?}

\question{The inequality $250 - 18d \ge 100$ models the number of liters of
fuel remaining in a generator after $d$ days of use, where the generator must
keep at least $100$ liters. What is the best interpretation of $18$ in this
context?
\choices{\item The generator uses $18$ liters of fuel each day.
\item The generator holds $18$ liters of fuel when it is full.
\item The generator can be used for at most $18$ days.
\item At least $18$ liters of fuel must remain at all times.}}

\question{What is the greatest value of $x$ that satisfies
\[ \dfrac{x + 2}{3} - \dfrac{x - 4}{2} \ge 1\,? \]
\choices{\item $6$ \item $8$ \item $10$ \item $16$}}

\qfr{The length of a rectangular garden is $3$ meters greater than its width
$w$ meters. The perimeter of the garden must be no more than $46$ meters.
What is the greatest possible value of $w$?}

\question{What is the greatest value of $x$ that satisfies
$-3(x - 4) \ge 2x + 7\,$?
\choices{\item $-5$ \item $-1$ \item $1$ \item $5$}}

\question{The boundary lines of the system
\[ \begin{cases} 3x + y \le 21 \\ y \ge x - 3 \end{cases} \]
intersect at the point $(x,y)$. What is the value of $y$?
\choices{\item $3$ \item $4$ \item $6$ \item $9$}}

\question{What is the greatest integer value of $x$ that satisfies
\[ 5 - \dfrac{x - 1}{4} > 2\,? \]
\choices{\item $11$ \item $12$ \item $13$ \item $14$}}

\qfr{In the given inequality, $c$ is a constant.
\[ 2x + c \le 5x - 4 \]
The solutions of the inequality are given by $x \ge 3$. What is the value of
$c$?}

\question{How many integer values of $x$ satisfy $-2 < 4 - 3x \le 13\,$?
\choices{\item $4$ \item $5$ \item $6$ \item $7$}}

\question{If $2 < x < 6$, what is the greatest integer value that the
expression $5 - 2x$ can have?
\choices{\item $-7$ \item $-1$ \item $0$ \item $1$}}

\qfr{A shop builds standard chairs and deluxe chairs. Each standard chair
requires $2$ hours of labor and each deluxe chair requires $5$ hours, and at
most $80$ hours of labor are available. The shop must build at least $10$
standard chairs. What is the greatest number of deluxe chairs the shop can
build?}

\question{The numbers $a$, $b$, and $c$ satisfy $a < b$ and $c < 0$. Which
statement must be true?
\choices{\item $ac > bc$ \item $ac < bc$ \item $a + c > b + c$
\item $\dfrac{a}{c} < \dfrac{b}{c}$}}

\question{In the given inequality, $m$ is a constant and $m < 0$.
\[ mx + 6 > 18 \]
Which inequality represents all solutions?
\choices{\item $x < \dfrac{12}{m}$ \item $x > \dfrac{12}{m}$
\item $x < \dfrac{24}{m}$ \item $x > \dfrac{24}{m}$}}

\qfr{A pair of numbers $(x,y)$ satisfies the system
\[ \begin{cases} y \le -2x + 14 \\ y \le 3x - 1 \\ y \ge 0 \end{cases} \]
What is the greatest possible value of $y$?}
