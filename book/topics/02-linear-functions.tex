\settopicname{Linear Functions}
\chapter{Linear Functions}
\topicopen{Algebra}{Evaluate a linear function, build one from two points or
from a table, and interpret its slope, its initial value, and a statement
such as $f(a)=b$ in the context the function models.}

\begin{essentials}
A linear function has the form $f(x)=mx+b$, where $m$ and $b$ are constants.

\textbf{The constant $b$} is $f(0)$: the output when the input is $0$. Its
graph crosses the $y$-axis at $(0,b)$, and in a context it is the starting
amount.

\textbf{The constant $m$} is the constant rate of change: the amount $f(x)$
changes each time $x$ increases by $1$. If $m>0$ the function increases; if
$m<0$ it decreases. From any two points,
\[ m=\dfrac{f(x_2)-f(x_1)}{x_2-x_1},\qquad b=f(x_1)-mx_1. \]
\textbf{From a table}, divide the change in $f(x)$ by the actual change in
$x$ --- not by the number of rows. A table is linear when equal steps in $x$
give equal steps in $f(x)$.

\textbf{Reading $f(a)=b$}: the input $a$ produces the output $b$. In context,
$a$ carries the input units and $b$ the output units; $m$ carries output
units per input unit.
\end{essentials}
