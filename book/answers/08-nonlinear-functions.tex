\akhead{Topic 8 — Nonlinear Functions}

\ans{1}{1}{A}{A product is zero when a factor is zero, so $x=3$ or $x=-5$.
Of the options, only $-5$ is a zero.}

\ans{1}{2}{$k=5$}{The $y$-intercept is $f(0)$, and
$f(0)=2(0)^2-7(0)+5=5$.}

\ans{1}{3}{C}{In vertex form $a(x-h)^2+k$ with $a=1>0$, the smallest value of
$(x-4)^2$ is $0$, so the minimum value of $f$ is $9$.}

\ans{1}{4}{B}{$f(0)=250(1.08)^0=250(1)=250$; the initial amount is the
coefficient in front of the power.}

\ans{1}{5}{$c=3$}{The zeros are $-2$ and $8$, and the axis of symmetry is
halfway between them: $x=\dfrac{-2+8}{2}=3$.}

\ans{1}{6}{A}{Adding a constant to the output raises every point, so
$y=f(x)+6$ is $y=f(x)$ shifted 6 units up.}

\ans{1}{7}{C}{$f(3)=6(2)^3=6\cdot 8=48$.}

\ans{1}{8}{$4$}{The factors vanish at $x=0$, $x=4$ and $x=-7$, so the
greatest $x$-coordinate is $4$.}

\ans{1}{9}{A}{Complete the square: $x^2-6x+11=(x^2-6x+9)+2=(x-3)^2+2$.}

\ans{1}{10}{A}{At $t=0$, $P(0)=1200(1.35)^0=1200$, so $1200$ is the number of
insects when the study began.}

\ans{1}{11}{$3$}{The vertex sits on the axis of symmetry:
$x=-\dfrac{b}{2a}=-\dfrac{-12}{2(2)}=3$.}

\ans{1}{12}{B}{The growth factor is $1.04$, so each year the value is
multiplied by $1.04$: an increase of $(1.04-1)\cdot 100\%=4\%$.}

\ans{1}{13}{A}{Replacing $x$ by $x+3$ makes the graph reach each output 3
units sooner, so the graph shifts 3 units left.}

\ans{1}{14}{$5$}{Factor: $x^2+3x-40=(x+8)(x-5)=0$, so $x=-8$ or $x=5$; the
positive solution is $5$.}

\ans{1}{15}{B}{For large $x$ the leading term $-2x^3$ dominates, and it is
negative for large positive $x$, so $p(x)$ decreases without bound.}

\ans{1}{16}{A}{$x^2-2x-24=(x-6)(x+4)$, and this factored form shows the
$x$-intercepts $6$ and $-4$ directly.}

\ans{1}{17}{$15$}{$3(2)^{t/5}=24$ gives $2^{t/5}=8=2^3$, so $\dfrac{t}{5}=3$
and $t=15$.}

\ans{1}{18}{B}{Exactly one $x$-intercept means the discriminant is zero:
$k^2-4(1)(36)=0$, so $k^2=144$ and, since $k>0$, $k=12$.}

\ans{1}{19}{A}{The zeros are $7$ and $-1$, so the vertex is at
$a=\dfrac{7+(-1)}{2}=3$, and $b=f(3)=(3-7)(3+1)=-16$.}

\ans{1}{20}{A}{Each time $t$ increases by 18 the exponent increases by 1 and
the mass is multiplied by $\tfrac{1}{2}$, so the mass halves every 18 days.}

\ans{1}{21}{D}{The axis of symmetry is $x=-\dfrac{b}{2a}=-\dfrac{b}{4}$.
Setting $-\dfrac{b}{4}=-3$ gives $b=12$.}

\ans{1}{22}{$a=-4$}{The zeros are $a$ and $6$, so $\dfrac{a+6}{2}=1$, giving
$a+6=2$ and $a=-4$.}

\ans{2}{1}{A}{Complete the square: $x^2+8x+3=(x^2+8x+16)-16+3=(x+4)^2-13$.}

\ans{2}{2}{$441$}{$B(2)=400(1.05)^2=400(1.1025)=441$ dollars.}

\ans{2}{3}{A}{The base of the exponential is the factor per hour, so each
hour the number of bacteria is multiplied by 3 --- it triples.}

\ans{2}{4}{C}{$x^3-9x=x(x-3)(x+3)$, so the zeros are $0$, $3$ and $-3$; the
greatest is $3$.}

\ans{2}{5}{$11$}{$f(4)=4^2-3(4)=4$, so $g(4)=f(4)+7=4+7=11$.}

\ans{2}{6}{A}{Negating the output sends $(x,y)$ to $(x,-y)$, which is a
reflection across the $x$-axis.}

\ans{2}{7}{A}{The base $0.88$ is the factor per year, so each year the value
is $0.88=88\%$ of the previous year's value (a $12\%$ decrease).}

\ans{2}{8}{$k=54$}{The vertex is at $x=-\dfrac{-14}{2}=7$, and
$f(7)=49-98+k=k-49$. Setting $k-49=5$ gives $k=54$.}

\ans{2}{9}{A}{Expand: $(2x-5)(x+3)=2x^2+6x-5x-15=2x^2+x-15$.}

\ans{2}{10}{A}{The leading term $3x^4$ has even degree and a positive
coefficient, so both ends of the graph rise without bound.}

\ans{2}{11}{$24$}{$8000(2)^{t/12}=32000$ gives $2^{t/12}=4=2^2$, so
$\dfrac{t}{12}=2$ and $t=24$.}

\ans{2}{12}{C}{Since $(3,0)$ is on the graph, $3^2+3b-15=0$, so $3b=6$ and
$b=2$. (Indeed $(x+5)(x-3)=x^2+2x-15$.)}

\ans{2}{13}{A}{$g(6)=f(6-4)=f(2)=2^2+1=5$.}

\ans{2}{14}{$64$}{The maximum is at $t=-\dfrac{64}{2(-16)}=2$, and
$h(2)=-16(4)+64(2)=64$ feet.}

\ans{2}{15}{A}{Since $a>0$ the parabola opens upward, so it has a minimum,
and it occurs halfway between the zeros $2$ and $6$, at $x=4$.}

\ans{2}{16}{C}{The vertex is at $x=-\dfrac{-16}{2(2)}=4$, and
$f(4)=32-64+k=k-32$. Setting $k-32=-6$ gives $k=26$.}

\ans{2}{17}{$a=2$}{Substitute the point: $a(1+3)(1-7)=-48$, so $-24a=-48$ and
$a=2$.}

\ans{2}{18}{A}{The exponent $t/4$ increases by 1 each time $t$ increases by
4, so the membership is multiplied by 1.5 every 4 months.}

\ans{2}{19}{A}{The factor $(x+2)^2$ has even multiplicity, so the graph
touches the $x$-axis at $x=-2$; $(x-5)$ has odd multiplicity, so the graph
crosses at $x=5$. (Note $p(0)=-20$, not $20$.)}

\ans{2}{20}{$2$}{The vertex of $h$ is at $x=\dfrac{8}{2}=4$, and $k(x)=h(x+2)$
shifts the graph 2 units left, so the vertex of $k$ is at $x=4-2=2$.}

\ans{2}{21}{A}{$f(x+1)=3(4)^{x+1}=3\cdot 4\cdot 4^x=12(4)^x$.}

\ans{2}{22}{B}{$f(x)=(x-3)^2+4$, so the minimum of $f$ is $4$. The graph of
$y=f(x)+k$ touches the $x$-axis exactly once when $4+k=0$, so $k=-4$.}
