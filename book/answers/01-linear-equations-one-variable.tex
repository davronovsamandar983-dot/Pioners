\akhead{Topic 1 --- Linear Equations in One Variable}

\ans{1}{1}{$x=7$}{Subtract $12$ from both sides: $5x=35$; divide by $5$ to
get $x=7$.}

\ans{1}{2}{C}{Divide both sides by $3$: $x-4=6$, so $x=10$. Choice B stops at
$x-4=6$; choice D forgets the final division.}

\ans{1}{3}{C}{Subtract $4x$: $3x-3=9$; add $3$: $3x=12$, so $x=4$. Choice A
comes from $3x=9-3$.}

\ans{1}{4}{$x=32$}{Subtract $3$: $\dfrac{x}{4}=8$; multiply by $4$ to get
$x=32$.}

\ans{1}{5}{B}{Subtract $12$: $-2x=18$; divide by $-2$ to get $x=-9$. Choice C
drops the negative sign.}

\ans{1}{6}{B}{Multiply both sides by $2$: $x+9=14$, so $x=5$. Choice A comes
from reading the left side as $\dfrac{x}{2}+9$.}

\ans{1}{7}{$12$}{The total is $40+15m=220$, so $15m=180$ and $m=12$ months.}

\ans{1}{8}{A}{Distribute: $8x+12=44$, so $8x=32$ and $x=4$. Choice C comes
from $8x=44+12$; choice B from dropping the $+12$.}

\ans{1}{9}{C}{Distribute both sides: $5x-10=3x+12$. Then $2x=22$, so $x=11$.
Choice B comes from failing to distribute one of the factors.}

\ans{1}{10}{$x=14$}{Multiply every term by $6$: $2(2x-1)=3(x+4)$, so
$4x-2=3x+12$ and $x=14$.}

\ans{1}{11}{B}{Model the amount left as $250-8t=90$. Then $8t=160$, so
$t=20$ minutes. Choice C comes from adding instead of subtracting.}

\ans{1}{12}{D}{Add $2.3$ and subtract $0.4x$: $3.5=0.5x$, so $x=7$. Choice C
comes from $2.3-1.2$ instead of $2.3+1.2$.}

\ans{1}{13}{A}{Subtract $2\ell$: $P-2\ell=2w$; divide by $2$ to get
$w=\dfrac{P-2\ell}{2}$. Choice B divides only the first term by $2$.}

\ans{1}{14}{$x=7$}{Distribute: $6x-15-4x+12=11$, so $2x-3=11$, $2x=14$, and
$x=7$.}

\ans{1}{15}{B}{The charge is $25+0.10t=43$, so $0.10t=18$ and $t=180$ text
messages. Choice A forgets to divide by $0.10$.}

\ans{1}{16}{B}{In $C=8n+150$ the coefficient of $n$ is the amount the cost
rises for each additional item, so $8$ is the cost in dollars per item; the
constant $150$ is the fixed cost.}

\ans{1}{17}{$x=20$}{Multiply every term by $10$: $5x-2x=60$, so $3x=60$ and
$x=20$.}

\ans{1}{18}{D}{Distributing gives $6x+15=6x+k$. The two sides are identical
for every $x$ exactly when $k=15$.}

\ans{1}{19}{C}{Rewrite as $(a-4)x=21$. There is no solution only when the
coefficient of $x$ is $0$ while the constant is not, so $a=4$.}

\ans{1}{20}{$x=5$}{The left side is $6x-1$ and the right side is $x+24$, so
$5x=25$ and $x=5$.}

\ans{1}{21}{A}{Multiply by $4$: $4y=3x-5$; add $5$: $4y+5=3x$; divide by $3$
to get $x=\dfrac{4y+5}{3}$. Choice B slips the sign on $5$.}

\ans{1}{22}{C}{Subtracting $5x$ leaves $c=-8$, which contains no $x$. If
$c=-8$ this is true for every $x$ (infinitely many solutions); otherwise it
is never true (no solution).}

\ans{2}{1}{A}{Cross multiply: $2(3x+1)=5(x-3)$, so $6x+2=5x-15$ and
$x=-17$. Choice B comes from not distributing the $5$ to the $-3$.}

\ans{2}{2}{$6$}{With width $w$, the length is $2w+3$, so
$2(2w+3)+2w=42$. Then $6w+6=42$, $6w=36$, and $w=6$ centimeters.}

\ans{2}{3}{C}{Distribute: $0.25x+2=0.5x-3$, so $5=0.25x$ and $x=20$. Choice
D comes from not distributing the $0.25$ to the $8$.}

\ans{2}{4}{B}{The fuel starts at $60$ liters and drops $9$ liters each hour,
so $f=60-9t$. Choice A uses the wrong sign for a decreasing amount.}

\ans{2}{5}{$x=8$}{Distribute: $5x-15-8+2x=3x+9$, so $7x-23=3x+9$, $4x=32$,
and $x=8$.}

\ans{2}{6}{A}{Subtract $32$: $F-32=\dfrac{9}{5}C$; multiply by $\dfrac{5}{9}$
to get $C=\dfrac{5(F-32)}{9}$. Choice D subtracts $32$ after dividing.}

\ans{2}{7}{B}{Distribute: $\dfrac{2}{3}x+4=\dfrac{1}{2}x+7$. Then
$\dfrac{1}{6}x=3$, so $x=18$. Choice A comes from not distributing the
$\dfrac{1}{2}$ to the $-4$.}

\ans{2}{8}{$300$}{The cost is $36+0.20m=96$, so $0.20m=60$ and $m=300$
miles.}

\ans{2}{9}{C}{Multiply every term by $4$: $2x+(x+2)=32$, so $3x=30$ and
$x=10$. Choice D forgets the final division by $3$.}

\ans{2}{10}{B}{Substitute $x=9$: $4(9+k)=6(9)-10=44$, so $9+k=11$ and $k=2$.
Choice C comes from distributing the $4$ only to the $x$.}

\ans{2}{11}{$10$}{Set the costs equal: $30+4c=50+2c$, so $2c=20$ and $c=10$
classes.}

\ans{2}{12}{C}{Set $5x-12=3x+4$. Then $2x=16$, so $x=8$. Choice A comes from
$2x=4-12$; choice D forgets to divide by $2$.}

\ans{2}{13}{C}{The number $x$ satisfies $x-0.4x=27$, so $0.6x=27$ and
$x=45$. Choice D divides by $0.4$ instead of $0.6$.}

\ans{2}{14}{$11$}{Substitute: $72=\dfrac{1}{2}(8)(7+b_2)=4(7+b_2)$, so
$18=7+b_2$ and $b_2=11$ centimeters.}

\ans{2}{15}{A}{Multiply both sides by $c$: $ax+b=cd$; subtract $b$ and
divide by $a$ to get $x=\dfrac{cd-b}{a}$. Choice C divides only $b$ by $a$.}

\ans{2}{16}{C}{The left side simplifies to $6x-11$. The equation is then
$6x-11=cx-11$, which is true for every $x$ exactly when $c=6$.}

\ans{2}{17}{B}{Rearranging gives $(k-6)x=3(k-6)$. If $k\neq6$ divide by
$k-6$ to get the single solution $x=3$; if $k=6$ the equation is $0=0$, so
every real number works.}

\ans{2}{18}{$6$}{Multiply by $3$: $ax-7=6x+15$. There is no solution only
when the $x$-coefficients match but the constants do not, so $a=6$ (and
$-7\neq15$).}

\ans{2}{19}{B}{Multiply every term by $6$: $2(x+4)-(x-2)=12$, so
$2x+8-x+2=12$, giving $x+10=12$ and $x=2$. Choice C drops the sign on
$-(-2)$.}

\ans{2}{20}{C}{Set the totals equal: $120+3t=40+5t$, so $80=2t$ and $t=40$
minutes. Choice D forgets to divide by $2$.}

\ans{2}{21}{$21$}{With $m=3$ the equation is $5(x-6)=3x+12$, so $5x-30=3x+12$,
$2x=42$, and $x=21$.}

\ans{2}{22}{D}{The left side is $\dfrac{x}{2}+\dfrac{b}{4}$, so the equation
is true for every $x$ exactly when $\dfrac{b}{4}=3$, that is $b=12$.}
